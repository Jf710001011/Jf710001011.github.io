📥 Input Format
  • The first line: an integer nnn — the number of positions to shift each character forward 1≤n≤1001 \leq n \leq 1001≤n≤100

📤 Output Format

  • Output one line — the encrypted string after applying the Caesar cipher.

    import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int n = in.nextInt();
        in.nextLine();
        StringBuilder sb = new StringBuilder(in.nextLine());
        // System.out.println((char)122);

        for (int i = 0; i < sb.length(); i++) {
            // Ascii 'z' = 122
            int offset = n % 26;
            // System.out.println(offset);

            // 0-26 represent a-z
            int codeFrom0 = sb.charAt(i) + offset - 97;
            // if over 26
            int effectIndex = codeFrom0 % 26;
            sb.setCharAt(i, (char) (effectIndex + 'a'));
        }

        System.out.println(sb.toString());
    }
}

public class QuickSort {

    /*
     * Each call to quickSort performs a partition to fix one pivot's position.
     */
    public static int partion(int[] arr, int low, int high) {

        // Choose the first element as the pivot
        int pivot = arr[low];

        // Preserve the original low and high bounds
        int i = low + 1, j = high;

        /*
         * Why use '<=': handles cases like [5, 4]
         */
        while (i <= j) {
            // From the front: find the first number greater than the pivot
            while (i <= j && arr[i] < pivot) {
                i++;
            }
            // From the back: find the first number less than the pivot
            while (i <= j && arr[j] > pivot) {
                j--;
            }

            if (i < j) {
                // Swap arr[i] and arr[j]
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }

        // Place the pivot in its correct position

        /*
         * Why use i - 1: to handle edge cases like [4, 5]
         */
        int temp = arr[i - 1];
        arr[low] = temp;
        arr[i - 1] = pivot;

        return i - 1;
    }

    /*
     * Three steps to process the array:
     * 1. Partition
     * 2. Recursively sort the left part
     * 3. Recursively sort the right part
     */
    public static void quickSort(int[] arr, int low, int high) {
        // Base case for recursion
        if (low <= high) {
            int pivotIndex = partion(arr, low, high);
            quickSort(arr, pivotIndex + 1, high);
            quickSort(arr, low, pivotIndex - 1);
        }
    }

void heapify(vector<int>& nums, int n, int i) {
    int indexLargest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    // 如果左子节点大于根节点
    if (left < n && nums[left] > nums[indexLargest]) {
        indexLargest = left;
    }

    // 如果右子节点大于根节点
    if (right < n && nums[right] > nums[indexLargest]) {

        heapify(nums, n, indexLargest);

    class Solution {
   private:
    vector<vector<int>> edges;
    vector<int> visit;
    bool finished = true;

   public:
    void dfs(int u) {
        // 标记为已被访问
        visit[u] = 1;

        // 访问u的邻接点
        for (int v : edges[u]) {
            // 有环（即不能完成所有课程）
            if (!finished) {
                return;
            }

            if (visit[v] == 0) {
                // 如果该邻接点未被访问过，则继续深度优先搜索
                dfs(v);
            } else if (visit[v] == 1) {
                // 如果该节点之前便处于搜索中，那么说明存在环
                finished = false;
                return;
            }
        }
        // 返回u,u及其邻接点搜索完成	
        visit[u] = 2;
    }
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        edges.resize(numCourses);
        visit.resize(numCourses);

        for (const auto& info : prerequisites) {
            edges[info[1]].push_back(info[0]);
        }

        for (int i = 0; i < numCourses && finished; ++i) {
            if (visit[i] == 0) {
                dfs(i);
            }
        }
        return finished;

struct DLinkedNode {
    int key, value;
    DLinkedNode* prev;
    DLinkedNode* next;
    DLinkedNode() : key(0), value(0), prev(nullptr), next(nullptr) {}
    DLinkedNode(int _key, int _value)

class LRUCache {
   private:
    unordered_map<int, DLinkedNode*> cache;
    DLinkedNode* head;
    // 移除尾部节点（被置换的节点）
    DLinkedNode* tail;
    // 比较size和capacity确定是否需要置换
    int size;
    int capacity;

   public:
    LRUCache(int _capacity) : capacity(_capacity), size(0) {
        head = new DLinkedNode();
        tail = new DLinkedNode();
        head->next = tail;
        tail->prev = head;
    }

    int get(int key) {
        if (!cache.count(key)) {
            return -1;
        }

        DLinkedNode* node = cache[key];
        // 刷到最新
        moveToHead(node);
        return node->value;
    }

    void put(int key, int value) {
        if (!cache.count(key)) {
            DLinkedNode* node = new DLinkedNode(key, value);
            cache[key] = node;
            // 刷新
            addToHead(node);
            ++size;

            // 超出容量
            if (size > capacity) {
                DLinkedNode* removed = removeTail();
                cache.erase(removed->key);
                delete removed;
                --size;
            }
        } else {
            DLinkedNode* node = cache[key];
            node->value = value;
            moveToHead(node);
        }
    }

    void addToHead(DLinkedNode* node) {
        node->prev = head;
        node->next = head->next;
        head->next = node;
        node->next->prev = node;
    }

    void removeNode(DLinkedNode* node) {
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }

    void moveToHead(DLinkedNode* node) {
        removeNode(node);
        addToHead(node);
    }

    DLinkedNode* removeTail() {
        DLinkedNode* node = tail->prev;
        removeNode(node);
        return node;
    }
};

   public:
    ListNode* detectCycle(ListNode* head) {
        // 从头节点出发：
        // 快指针走2步，慢指针走一步。
        // 重合需要满足的要求：快指针追上慢指针（在环内），快指针刚好超过慢指针n圈环
        ListNode* fast = head;
        ListNode* slow = head;

        while (true) {
            if (!fast || !(fast->next)) {
                return nullptr;
            }

            fast = fast->next->next;
            slow = slow->next;
            if (fast == slow) {
                // f（快指针步数）；s(慢指针步数)；b(环的步数)
                // 在第一次相遇时满足：
                // 1. f = 2s; (2倍速度)
                // 2. f = s+nb
                // 由以上两式知s=nb
                // 画个示意图：假设快慢指针在环内的同一点即s+(n-1)b处，则再次相遇快指针要超过慢指针一圈
                break;
            }
        }
        // a(链表头结点到环起点的步数)；x（从环起点到当前位置的步数）;k<n(绕环的圈数)
        // 1. s = nb
        // 2. s = （n-1）b+a+x
        // 因此慢指针在此时的位置（第一次相遇）再走a步必定在环的起始处，
        // 如何确定走a步：让另一指针在头节点出发，该指针与慢指针相遇则确定是走了a步
        fast = head;
        while (slow != fast) {
            fast = fast->next;
            slow = slow->next;
        }
        return slow;
    }
};

        // 自定义对 array<int, 26> 类型的哈希函数
        auto arrayHash = [fn = hash<int>{}] (const array<int, 26>& arr) -> size_t {
            return accumulate(arr.begin(), arr.end(), 0u, [&](size_t acc, int num) {
                return (acc << 1) ^ fn(num);
            });
        };

        unordered_map<array<int, 26>, vector<string>, decltype(arrayHash)> mp(0, arrayHash);

            int length = str.length();
            for (int i = 0; i < length; ++i) {
                counts[str[i] - 'a'] ++;
            }
            mp[counts].emplace_back(str);
        }
        vector<vector<string>> ans;
        for (auto it = mp.begin(); it != mp.end(); ++it) {
            ans.emplace_back(it->second);
        }
        return ans;
    }

class Solution {
   public:
    int search(vector<int>& nums, int target) {
        int left = 0;
        int right = nums.size() - 1;

        // 总的来说比起常规
        while (left < right) {
            int mid = (left + right) >> 1;
            if (nums[mid] == target) {
                return mid;
            }

            if (nums[left] <= nums[mid]) {    // 前半段有序
                if (target >= nums[left] && target < nums[mid]) {    // 先查找前半段;注意等号
                    right = mid - 1;
                } else {    // 在后半段查找
                    left = mid + 1;
                }
            } else {    // 后半段有序
                if(target > nums[mid] && target <= nums[right]) {    // 先查找后半段
                    left = mid + 1;
                } else {    // 在前半段查找
                    right = mid - 1;
                }

            }
        }
        return nums[left] == target ? left : -1;
    }
};

        }
    }

            return ans;
        }

class Solution {
   public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        sort(nums.begin(), nums.end());    // 先排序
        vector<vector<int>> ans;
        int left = 0;
        int mid;
        int right = nums.size() - 1;
        if (nums[left] > 0 || nums[right] < 0) {
            return ans;
        }

        // 每次固定左指针，最中、右指针处理
        for (left; left < nums.size() - 2; left++) {
            if(left != 0 && nums[left] == nums[left-1]) {   // left 也得去重
                continue;   // 不是break
            }

            // left 移动，则mid,right重置
            mid = left + 1;
            right = nums.size() - 1;

            while (mid < right) {
                if (nums[left] + nums[mid] + nums[right] > 0) {
                    right--;
                } else if (nums[right] + nums[mid] + nums[left] < 0) {
                    mid++;
                } else {
                    ans.push_back(vector<int>{nums[left], nums[mid], nums[right]});

                    //  去重
                    // mid 先去重
                    mid++;
                    while (mid < right && nums[mid] == nums[mid - 1]) {
                        mid++;
                    }

                    right--;
                    while (mid < right && nums[right] == nums[right + 1]) {
                        right--;
                    }

                }
            }
        }
        return ans;
    }
};

    • 两个边界肯定不是同时移动，那么就是一次移动一个
    • 每次不管左移还是右移，底边都是-1，因此得确保高尽量高，也就是长轴保持，移动短轴寻找更长轴
    • 不容易注意到的一个问题是：如果要保证之后的结果比现在更大，那么最短轴一定要更长（因为底边再减小），也就是说每次移动，最短轴一定要变化。如果最短轴不变，那么这次移动是无效的。

class Solution {
   public:
    int maxArea(vector<int>& height) {
        int left = 0;
        int right = height.size() - 1;
        int ans = 0;
        while (left < right) {
            ans = max(ans, (right - left) * min(height[left], height[right]));
            if (height[left] < height[right]) {
                left++;
            } else {
                right--;
            }
        }
        return ans;
    }
};

class Solution {

    int minimumLength(string s) {    // 双指针法
        if (s.length() <= 1) {
            return s.length();
        }
        int left = 0;
        int right = s.length() - 1;
        while (left < right && s[left] == s[right]) {
                left++;
                while (left < right && s[left - 1] == s[left]) {    // 删除左边的全部
                    left++;
                }
                right--;
                while (left < right && s[right + 1] == s[right]) {    // 删除右边的全部
                    right--;
                }
            }
            return right - left + 1;
